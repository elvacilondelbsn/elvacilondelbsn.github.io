}

async function handleLogin(env, req) {
  let body;
  try {
    body = await req.json();
  } catch {
    return json({ ok: false, error: "JSON inválido" }, 400);
  }
  const username = String(body.username || "").trim();
  const password = String(body.password || "");
  if (!username || !password) return json({ ok: false, error: "Faltan credenciales" }, 400);

  const user = await env.DB.prepare(
    "SELECT id, password_hash, salt FROM admin_users WHERE username = ?"
  ).bind(username).first();
  if (!user) return json({ ok: false, error: "Credenciales inválidas" }, 401);

  const hash = await sha256Hex(user.salt + ":" + password);
  if (hash !== user.password_hash) {
    return json({ ok: false, error: "Credenciales inválidas" }, 401);
  }

  const token = randomHex(32);
  const expires_at = Date.now() + TOKEN_TTL_MS;
  await env.DB.prepare(
    "INSERT INTO sessions (token, user_id, expires_at) VALUES (?, ?, ?)"
  ).bind(token, user.id, expires_at).run();

  return json({ ok: true, token, expires_at });
}

async function handlePublicContent(env, url) {
  const section = url.searchParams.get("section") || "";
  if (!(await sectionExists(env, section))) {
    return json({ ok: false, error: "Sección inválida" }, 400);
  }
  const rows = await env.DB.prepare(
    "SELECT id, section, title, subtitle, body, image_url, meta, sort_order, updated_at FROM content WHERE section = ? AND is_published = 1 ORDER BY sort_order ASC, id DESC"
  ).bind(section).all();
  return json({ ok: true, section, items: rows.results || [] });
}

async function handleAdminList(env, url) {

    rows = await env.DB.prepare(
      "SELECT * FROM content WHERE section = ? ORDER BY sort_order ASC, id DESC"
    ).bind(section).all();
  } else {
    rows = await env.DB.prepare(
      "SELECT * FROM content ORDER BY section ASC, sort_order ASC, id DESC"
    ).all();
  }
  return json({ ok: true, items: rows.results || [] });
}

async function handleCreate(env, req) {
  let body;
  try {
    body = await req.json();
  } catch {
    return json({ ok: false, error: "JSON inválido" }, 400);
  }
  const section = String(body.section || "");
  const title = String(body.title || "").trim();
  if (!(await sectionExists(env, section))) return json({ ok: false, error: "Sección inválida" }, 400);
  if (!title) return json({ ok: false, error: "El título es obligatorio" }, 400);

  const ts = nowIso();
  const res = await env.DB.prepare(
    `INSERT INTO content (section, title, subtitle, body, image_url, meta, sort_order, is_published, created_at, updated_at)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`
  ).bind(
    section,
    title,
    String(body.subtitle || ""),
    String(body.body || ""),
    String(body.image_url || ""),
    typeof body.meta === "string" ? body.meta : JSON.stringify(body.meta || {}),
    parseInt(body.sort_order || "0", 10) || 0,
    body.is_published === false || body.is_published === 0 ? 0 : 1,
    ts,
    ts
  ).run();
  return json({ ok: true, id: res.meta.last_row_id });
}

async function handleUpdate(env, req, id) {
  let body;
  try {
    body = await req.json();
  } catch {
    return json({ ok: false, error: "JSON inválido" }, 400);
  }
  const fields = [];
  const vals = [];
  for (const key of ["title", "subtitle", "body", "image_url", "sort_order", "is_published"]) {
    if (body[key] !== undefined) {
      fields.push(`${key} = ?`);
      if (key === "sort_order") vals.push(parseInt(body[key] || "0", 10) || 0);
      else if (key === "is_published") vals.push(body[key] ? 1 : 0);
      else vals.push(String(body[key]));
    }
  }
  if (body.meta !== undefined) {
    fields.push("meta = ?");
    vals.push(typeof body.meta === "string" ? body.meta : JSON.stringify(body.meta));
  }
  if (body.section !== undefined && (await sectionExists(env, String(body.section)))) {
    fields.push("section = ?");
    vals.push(String(body.section));
  }
  if (!fields.length) return json({ ok: false, error: "Nada que actualizar" }, 400);
  fields.push("updated_at = ?");
  vals.push(nowIso());
  vals.push(id);
  await env.DB.prepare(`UPDATE content SET ${fields.join(", ")} WHERE id = ?`).bind(...vals).run();
  return json({ ok: true });
}

async function handleDelete(env, id) {
  const row = await env.DB.prepare("SELECT image_url FROM content WHERE id = ?").bind(id).first();
  await env.DB.prepare("DELETE FROM content WHERE id = ?").bind(id).run();
  if (row) await deleteFotoIfOurs(env, row.image_url);
  return json({ ok: true });
}

async function handlePublicCustomSections(env) {
  const secs = await env.DB.prepare(
    "SELECT key, title FROM custom_sections ORDER BY sort_order ASC, title ASC"
  ).all();
  const out = [];
  for (const s of (secs.results || [])) {
    const c = await env.DB.prepare(
      "SELECT COUNT(*) AS n FROM content WHERE section = ? AND is_published = 1"
    ).bind(s.key).first();
    if (c && c.n > 0) out.push({ key: s.key, title: s.title });
  }
  return json({ ok: true, sections: out });
}
