/**
 * VACILON-API — Backend de EL VACILÓN DEL BSN
 * Cloudflare Worker + D1 + KV (sin tarjeta, todo en plan gratuito)
 *
 * Bindings requeridos en el Worker:
 *   D1 variable "DB"   -> base de datos "vacilon-bsn"
 *   KV variable "FOTOS" -> namespace "vacilon-fotos"
 *
 * Endpoints públicos:
 *   GET  /api/content?section=noticias|cambios|firmas|agencia_libre|franquicias|<seccion-personalizada>
 *   GET  /api/custom-sections          (secciones personalizadas con contenido publicado)
 *   GET  /api/fotos/:key                    (sirve foto desde KV)
 *
 * Endpoints admin (requieren header Authorization: Bearer <token>):
 *   POST /api/login            {username, password} -> {token}
 *   GET  /api/admin/content     (todo, incluyendo no publicado)
 *   POST /api/content          (crear)
 *   PUT  /api/content/:id      (editar)
 *   DELETE /api/content/:id    (borrar)
 *   POST /api/upload           (multipart, campo "file" -> {url})
 *   GET  /api/admin/custom-sections            (listar secciones personalizadas)
 *   POST /api/admin/custom-sections   {title} -> {key, title}
 *   DELETE /api/admin/custom-sections/:key    (borra sección y su contenido)
 *
 * Configuración inicial (solo si aún no hay usuarios admin):
 *   GET  /api/setup-needed     -> {needed: true|false}
 *   POST /api/setup            {username, password, key} -> {ok:true}
 *   La llave SETUP_KEY se define al desplegar y se entrega al dueño.
 */

const SECTIONS = ["noticias", "cambios", "firmas", "agencia_libre", "franquicias"];

// Secciones personalizadas creadas desde el panel (tabla custom_sections).
async function sectionExists(env, s) {
  if (SECTIONS.includes(s)) return true;
  if (!s || !/^[a-z0-9-]{1,40}$/.test(s)) return false;
  const r = await env.DB.prepare("SELECT 1 FROM custom_sections WHERE key = ?").bind(s).first();
  return !!r;
}

function slugify(t) {
  const s = String(t || "").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "").slice(0, 40);
  return s || "seccion";
}

// ---------- utilidades ----------

function json(data, status = 200, extraHeaders = {}) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      "content-type": "application/json; charset=utf-8",
      ...extraHeaders,
    },
  });
}

function corsHeaders(origin) {
  const allowed = [
    "https://elvacilondelbsn.github.io",
    "http://127.0.0.1:8901",
    "http://localhost:8901",
  ];
  const allow = allowed.includes(origin) ? origin : allowed[0];
  return {
    "access-control-allow-origin": allow,
    "access-control-allow-methods": "GET, POST, PUT, DELETE, OPTIONS",
    "access-control-allow-headers": "content-type, authorization",
    "access-control-max-age": "86400",
  };
}

function withCors(res, origin) {
  const h = new Headers(res.headers);
  const c = corsHeaders(origin);
  for (const k of Object.keys(c)) h.set(k, c[k]);
  return new Response(res.body, { status: res.status, headers: h });
}

async function sha256Hex(text) {
  const buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(text));
  return [...new Uint8Array(buf)].map((b) => b.toString(16).padStart(2, "0")).join("");
}

function randomHex(nBytes) {
  const b = new Uint8Array(nBytes);
  crypto.getRandomValues(b);
  return [...b].map((x) => x.toString(16).padStart(2, "0")).join("");
}

function nowIso() {
  return new Date().toISOString();
}

async function getSessionUser(env, req) {
  const auth = req.headers.get("authorization") || "";
  const m = auth.match(/^Bearer\s+(.+)$/i);
  if (!m) return null;
  const token = m[1].trim();
  const row = await env.DB.prepare(
    "SELECT user_id, expires_at FROM sessions WHERE token = ?"
  ).bind(token).first();
  if (!row) return null;
  if (row.expires_at < Date.now()) {
    await env.DB.prepare("DELETE FROM sessions WHERE token = ?").bind(token).run();
    return null;
  }
  return row.user_id;
}

// ---------- handlers ----------

async function handleSetupNeeded(env) {
  const row = await env.DB.prepare("SELECT COUNT(*) AS n FROM admin_users").first();
  return json({ ok: true, needed: !row || row.n === 0 });
}

async function handleSetup(env, req) {
  const row = await env.DB.prepare("SELECT COUNT(*) AS n FROM admin_users").first();
  if (row && row.n > 0) {
    return json({ ok: false, error: "El panel ya está configurado" }, 403);
  }
  let body;
  try {
    body = await req.json();
  } catch {
    return json({ ok: false, error: "JSON inválido" }, 400);
  }
  if (String(body.key || "") !== SETUP_KEY || SETUP_KEY === "REEMPLAZAR_AL_DESPLEGAR") {
    return json({ ok: false, error: "Llave de configuración inválida" }, 403);
  }
  const username = String(body.username || "").trim();
  const password = String(body.password || "");
  if (username.length < 3) return json({ ok: false, error: "El usuario debe tener al menos 3 letras" }, 400);
  if (password.length < 8) return json({ ok: false, error: "La contraseña debe tener al menos 8 caracteres" }, 400);
  const salt = randomHex(16);
  const password_hash = await sha256Hex(salt + ":" + password);
  try {
    await env.DB.prepare(
      "INSERT INTO admin_users (username, password_hash, salt, created_at) VALUES (?, ?, ?, ?)"
    ).bind(username, password_hash, salt, nowIso()).run();
  } catch {
    return json({ ok: false, error: "Ese usuario ya existe" }, 400);
  }
  return json({ ok: true });
