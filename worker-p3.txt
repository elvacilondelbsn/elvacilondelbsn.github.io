}

async function handleCreateCustomSection(env, req) {
  let body;
  try {
    body = await req.json();
  } catch {
    return json({ ok: false, error: "JSON inválido" }, 400);
  }
  const title = String(body.title || "").trim();
  if (!title) return json({ ok: false, error: "El nombre es obligatorio" }, 400);
  const key = slugify(title);
  if (SECTIONS.includes(key)) return json({ ok: false, error: "Ese nombre está reservado" }, 400);
  const ex = await env.DB.prepare("SELECT 1 FROM custom_sections WHERE key = ?").bind(key).first();
  if (ex) return json({ ok: false, error: "Ya existe una sección con ese nombre" }, 400);
  const mx = await env.DB.prepare("SELECT COALESCE(MAX(sort_order),0) AS m FROM custom_sections").first();
  await env.DB.prepare(
    "INSERT INTO custom_sections (key, title, sort_order, created_at) VALUES (?, ?, ?, ?)"
  ).bind(key, title, ((mx && mx.m) || 0) + 1, nowIso()).run();
  return json({ ok: true, key, title });
}

async function handleDeleteCustomSection(env, key) {
  if (!/^[a-z0-9-]{1,40}$/.test(key || "")) return json({ ok: false, error: "No encontrada" }, 404);
  const ex = await env.DB.prepare("SELECT 1 FROM custom_sections WHERE key = ?").bind(key).first();
  if (!ex) return json({ ok: false, error: "No encontrada" }, 404);
  const items = await env.DB.prepare("SELECT image_url FROM content WHERE section = ?").bind(key).all();
  for (const it of (items.results || [])) await deleteFotoIfOurs(env, it.image_url);
  await env.DB.prepare("DELETE FROM content WHERE section = ?").bind(key).run();
  await env.DB.prepare("DELETE FROM custom_sections WHERE key = ?").bind(key).run();
  return json({ ok: true });
}

async function handleUpload(env, req) {
  let form;
  try {
    form = await req.formData();
  } catch {
    return json({ ok: false, error: "Formulario inválido" }, 400);
  }
  const file = form.get("file");
  if (!file || typeof file.arrayBuffer !== "function") {
    return json({ ok: false, error: "No se recibió el archivo" }, 400);
  }
  if (file.size > MAX_UPLOAD_BYTES) {
    return json({ ok: false, error: "La foto no puede pasar de 5 MB" }, 400);
  }
  const type = (file.type || "").toLowerCase();
  if (!type.startsWith("image/")) {
    return json({ ok: false, error: "Solo se permiten fotos (JPG, PNG, WebP)" }, 400);
  }
  const ext = type.includes("png") ? "png" : type.includes("webp") ? "webp" : "jpg";
  const key = `${Date.now()}-${randomHex(8)}.${ext}`;
  const buf = await file.arrayBuffer();
  await env.FOTOS.put(key, buf, { metadata: { contentType: type } });
  return json({ ok: true, url: `/api/fotos/${key}`, key });
}

async function handleFoto(env, key) {
  if (!key || /[^a-zA-Z0-9._-]/.test(key)) return new Response("No encontrado", { status: 404 });
  const { value, metadata } = await env.FOTOS.getWithMetadata(key, { type: "arrayBuffer" });
  if (!value) return new Response("No encontrado", { status: 404 });
  const headers = new Headers();
  headers.set("content-type", (metadata && metadata.contentType) || "image/jpeg");
  headers.set("cache-control", "public, max-age=31536000, immutable");
  return new Response(value, { headers });
}

async function deleteFotoIfOurs(env, image_url) {
  const m = String(image_url || "").match(/^\/api\/fotos\/([a-zA-Z0-9._-]+)$/);
  if (m) {
    try { await env.FOTOS.delete(m[1]); } catch { /* no crítico */ }
  }
}

// ---------- router ----------

export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    const origin = req.headers.get("origin") || "";
    const path = url.pathname;

    if (req.method === "OPTIONS") {
      return new Response(null, { status: 204, headers: corsHeaders(origin) });
    }

    let res;
    try {
      if (path === "/api/login" && req.method === "POST") {
        res = await handleLogin(env, req);
      } else if (path === "/api/setup-needed" && req.method === "GET") {
        res = await handleSetupNeeded(env);
      } else if (path === "/api/setup" && req.method === "POST") {
        res = await handleSetup(env, req);
      } else if (path === "/api/content" && req.method === "GET") {
        res = await handlePublicContent(env, url);
      } else if (path === "/api/custom-sections" && req.method === "GET") {
        res = await handlePublicCustomSections(env);
      } else if (path === "/api/fotos/" || path.startsWith("/api/fotos/")) {
        const key = path.slice("/api/fotos/".length);
        res = await handleFoto(env, key);
        return withCors(res, origin);
      } else if (path.startsWith("/api/admin") || (path.startsWith("/api/content/") && req.method !== "GET") || (path === "/api/content" && req.method === "POST") || path === "/api/upload") {
        const userId = await getSessionUser(env, req);
        if (!userId) {
          res = json({ ok: false, error: "No autorizado" }, 401);
        } else if (path === "/api/admin/content" && req.method === "GET") {
          res = await handleAdminList(env, url);
        } else if (path === "/api/content" && req.method === "POST") {
          res = await handleCreate(env, req);
        } else if (path === "/api/upload" && req.method === "POST") {
          res = await handleUpload(env, req);
        } else if (path === "/api/admin/custom-sections" && req.method === "GET") {
          res = await handleListCustomSections(env);
        } else if (path === "/api/admin/custom-sections" && req.method === "POST") {
          res = await handleCreateCustomSection(env, req);
        } else {
          const mcs = path.match(/^\/api\/admin\/custom-sections\/([a-z0-9-]+)$/);
          const m = path.match(/^\/api\/content\/(\d+)$/);
          if (mcs && req.method === "DELETE") res = await handleDeleteCustomSection(env, mcs[1]);
          else if (m && req.method === "PUT") res = await handleUpdate(env, req, m[1]);
          else if (m && req.method === "DELETE") res = await handleDelete(env, m[1]);
          else res = json({ ok: false, error: "No encontrado" }, 404);
        }
      } else if (path === "/" || path === "/api") {
        res = json({ ok: true, service: "vacilon-api", version: 1 });
      } else {
        res = json({ ok: false, error: "No encontrado" }, 404);
      }
    } catch (e) {
      res = json({ ok: false, error: "Error interno" }, 500);
    }
    return withCors(res, origin);
  },
};
